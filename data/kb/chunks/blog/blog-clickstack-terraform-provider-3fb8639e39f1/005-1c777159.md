```
1variable "logs_source_id" {
2  description = "ID of the ClickStack logs source used by the dashboard"
3  type        = string
4}
5
6resource "clickhouse_clickstack_dashboard" "simple_logs" {
7  dashboard_json = jsonencode({
8    name    = "simple logs dashboard"
9
10    tiles = [
11      {
12        name = "Log count over time by service",
13        id = "9kcn995dbkdxjlw3jj28k"
14        x  = 0
15        y  = 0
16        w  = 24
17        h  = 11
18
19        config = {
20          name                        = "Logs over time"
21          sourceId                    = var.logs_source_id
22          displayType                 = "line"
23          granularity                 = "auto"
24          alignDateRangeToGranularity = true
25
26          select = [
27            {
28              aggFn                = "count"
29              aggCondition         = ""
30              aggConditionLanguage = "lucene"
31              valueExpression      = ""
32            }
33          ]
34
35          where         = ""
36          whereLanguage = "lucene"
37          groupBy       = "ServiceName"
38        }
39      }
40    ]
41
42    filters    = []
43    containers = []
44  })
45}
46
47output "dashboard_id" {
48  value = clickhouse_clickstack_dashboard.simple_logs.id
49}
```
Copy command
The `dashboard_json` value maps to the ClickStack v2 dashboard API. Keeping it inside `jsonencode` lets you use Terraform variables without constructing JSON strings by hand.
