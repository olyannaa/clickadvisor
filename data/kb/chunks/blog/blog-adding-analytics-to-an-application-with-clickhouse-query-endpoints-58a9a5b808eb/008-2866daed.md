trimmed down version of the application where users can enter a Python package and Github stats are rendered. To render a few details about the package, along with Github statistics, our query returns columns from the projects table:

```
1SET param_package_name='boto3'
2
3WITH
4  (
5        SELECT version
6        FROM pypi.projects
7        WHERE name = {package_name:String}
8        ORDER BY arrayMap(x -> toUInt8OrDefault(x, 0), splitByChar('.', version)) DESC
9        LIMIT 1
10  ) AS max_version,
11  project_details AS (
12     SELECT
13        name,
14        max_version,
15        summary,
16        author,
17        author_email,
18        license,
19        home_page,
20        trim(TRAILING '/' FROM regexpExtract(arrayFilter(l -> (l LIKE '%https://github.com/%'), arrayConcat(project_urls, [home_page]))[1], '.*https://github.com/(.*)')) AS github
21     FROM pypi.projects
22     WHERE (name = {package_name:String})
23     ORDER BY upload_time DESC
24     LIMIT 1
25  ),
26  id AS (
27      SELECT repo_id
28      FROM github.repo_name_to_id
29      WHERE repo_name IN (SELECT github FROM project_details) LIMIT 1
30  ),
31  stats AS (
32     SELECT
33        uniqExactIf(actor_login, (event_type = 'WatchEvent') AND (action = 'started')) AS stars,
34        uniqExactIf(number, event_type = 'IssuesEvent') AS issues,
35        uniqExactIf(actor_login, event_type = 'ForkEvent') AS forks,
36        uniqExactIf(number, event_type = 'PullRequestEvent') AS prs
37     FROM github.github_events_v2
38     WHERE (repo_id IN id)
39  )
40 SELECT * FROM project_details, stats FORMAT Vertical
```
