```
1async def handle_slack_query(event, say):
2    user = event["user"]
3    text = event.get("text", "")
4    thread_ts = event.get("thread_ts") or event["ts"]
5    channel = event["channel"]
6
7    await say(text=f"<@{user}>: Let me think...", thread_ts=thread_ts)
8
9    async def do_agent():
10        # Build context from thread if present
11        context = ""
12        if thread_ts and thread_ts != event["ts"]:
13            client = AsyncWebClient(token=SLACK_BOT_TOKEN)
14            replies = await client.conversations_replies(channel=channel, ts=thread_ts)
15            # Exclude the current message, and bot messages
16            messages = [m for m in replies["messages"] if m["ts"] != event["ts"]]
17            # Format as "user: message"
18            context_lines = []
19            for m in messages:
20                uname = m.get("user", "bot")
21                msg = m.get("text", "")
22                context_lines.append(f"{uname}: {msg}")
23            context = "n".join(context_lines)
24
25        # Compose prompt for the agent
26        if context:
27            prompt = f"Thread context so far:n{context}nnNew question: {text}"
28        else:
29            prompt = text
30
31        async with agent.run_mcp_servers():
32            result = await agent.run(prompt)
33            await say(text=f"{result.output}", thread_ts=thread_ts)
34
35    asyncio.create_task(do_agent())
```
