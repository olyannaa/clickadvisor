2	track_genre,
3	sparkbar(40)(CAST(duration_ms, 'UInt32'), c) AS distribution

5(
6	SELECT
7    	track_genre,

9    	duration_ms
10	FROM spotify

12    	track_genre,
13    	round(duration_ms, -4) AS duration_ms
14	ORDER BY duration_ms ASC
