on the runner, we use the [`hashicorp/setup-terraform`](https://github.com/hashicorp/setup-terraform) action. This installs Terraform on the Github actions CLI runner and exposes its CLI so we can make calls like we would from a terminal. Our final workflow is shown below:

```
1integration-tests-cloud:
2  runs-on: ubuntu-latest
3  defaults:
4	run:
5  	shell: bash
6  strategy:
7	max-parallel: 1
8	fail-fast: true
9	matrix:
10  	go:
11    	- "1.19"
12    	- "1.20"
13  steps:
14	- name: Check Out Code
15  	uses: actions/checkout@v3
16
17	- name: Setup Terraform
18  	uses: hashicorp/setup-terraform@v2.0.3
19  	with:
20    	terraform_version: 1.3.4
21    	terraform_wrapper: false
22
23	- name: Terraform Init
24  	id: init
25  	run: terraform init
26
27	- name: Terraform Validate
28  	id: validate
29  	run: terraform validate -no-color
30
31	- name: Set Service Name
32  	run: echo "TF_VAR_service_name=go_client_tests_$(date +'%Y_%m_%d_%H_%M_%S')" >> $GITHUB_ENV
33
34	- name: Terraform Apply
35  	id: apply
36  	run: terraform apply -no-color -auto-approve
37  	env:
38    	TF_VAR_organization_id: ${{ secrets.INTEGRATIONS_TEAM_TESTS_ORGANIZATION_ID }}
39    	TF_VAR_token_key:  ${{ secrets.INTEGRATIONS_TEAM_TESTS_TOKEN_KEY }}
40    	TF_VAR_token_secret:  ${{ secrets.INTEGRATIONS_TEAM_TESTS_TOKEN_SECRET }}
41    	TF_VAR_service_password: ${{ secrets.INTEGRATIONS_TEAM_TESTS_CLOUD_PASSWORD }}
42
43	- name: Set Host
44  	run: echo "CLICKHOUSE_HOST=$(terraform output -raw CLICKHOUSE_HOST)" >> $GITHUB_ENV
45
46	- name: Service Id
47  	run: terraform output -raw SERVICE_ID
48
49	- name: Install Go ${{ matrix.go }}
50  	uses: actions/setup-go@v2.1.5
51  	with:
52    	stable: false
53    	go-version: ${{ matrix.go }}
54
55	- name: Run tests
56  	env:
57    	CLICKHOUSE_PASSWORD: ${{ secrets.INTEGRATIONS_TEAM_TESTS_CLOUD_PASSWORD }}
58    	CLICKHOUSE_USE_DOCKER: false
59    	CLICKHOUSE_USE_SSL: true
60  	run: |
61    	CLICKHOUSE_DIAL_TIMEOUT=20 CLICKHOUSE_TEST_TIMEOUT=600s CLICKHOUSE_QUORUM_INSERT=2 make test
62
63	- name: Cleanup
64  	if: always()
65  	run: terraform destroy -no-color -auto-approve
66  	env:
67    	TF_VAR_organization_id: ${{ secrets.INTEGRATIONS_TEAM_TESTS_ORGANIZATION_ID }}
68    	TF_VAR_token_key:  ${{ secrets.INTEGRATIONS_TEAM_TESTS_TOKEN_KEY }}
69    	TF_VAR_token_secret:  ${{ secrets.INTEGRATIONS_TEAM_TESTS_TOKEN_SECRET }}
70    	TF_VAR_service_password: ${{ secrets.INTEGRATIONS_TEAM_TESTS_CLOUD_PASSWORD }}
```
