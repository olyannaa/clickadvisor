```
1require 'adbc'
2require 'arrow'
3
4# Schema mirrors the page_view_events table columns
5schema = Arrow::Schema.new([
6  Arrow::Field.new("user_id",    Arrow::StringDataType.new),
7  Arrow::Field.new("page_views", Arrow::UInt32DataType.new),
8  Arrow::Field.new("event_time", Arrow::TimestampDataType.new(:second))
9])
10
11database = ADBC::Database.new
12
13begin
14  database.set_option("driver", "clickhouse")
15  database.set_option("uri", "clickhouse://your-host.clickhouse.cloud:8443")
16  database.set_option("username", "your_username")
17  database.set_option("password", "your_password")
18  database.set_load_flags(ADBC::LoadFlags::DEFAULT)
19  database.init
20
21  database.connect do |conn|
22    # Three batches of page view events arriving from the background job
23    incoming = [
24      { users: ["Mark",  "Luke",  "Jim"],  page_views: [12, 45, 8]  },
25      { users: ["Ian",   "Peter"],         page_views: [33, 7]       },
26      { users: ["Mark",  "Eva"],           page_views: [19, 62]      }
27    ]
28
29    # Build a RecordBatch per batch. Data stays columnar throughout.
30    batches = incoming.map do |data|
31      Arrow::RecordBatch.new(schema, data[:users].size, [
32        Arrow::StringArray.new(data[:users]),
33        Arrow::UInt32Array.new(data[:page_views]),
34        Arrow::TimestampArray.new(:second, Array.new(data[:users].size, Time.now.to_i))
35      ])
36    end
37
38    # Wrap all batches in a reader. ADBC streams them as one operation.
39    reader = Arrow::RecordBatchReader.new(batches, schema)
40
41    conn.open_statement do |statement|
42      statement.ingest("page_view_events", reader, mode: :append)
43    end
44  end
45ensure
46  database.release
47end
```

### Reading data \#

Querying `page_view_events` with a GROUP BY returns an Arrow table of totals per user.

```
1require 'adbc'
2
3database = ADBC::Database.new
4
5begin
6  database.set_option("driver", "clickhouse")
7  database.set_option("uri", "clickhouse://your-host.clickhouse.cloud:8443")
8  database.set_option("username", "your_username")
9  database.set_option("password", "your_password")
10  database.set_load_flags(ADBC::LoadFlags::DEFAULT)
11  database.init
12
13  database.connect do |connection|
14    # In production, query user_activity (SummingMergeTree) for pre-merged totals.
15    # sum() handles recently inserted rows not yet merged by ClickHouse.
16    activity, = connection.query(<<~SQL)
17      SELECT
18          user_id,
19          sum(page_views) AS total_page_views
20      FROM page_view_events
21      GROUP BY user_id
22      ORDER BY total_page_views DESC
23      LIMIT 10
24    SQL
25
26    puts activity
27  end
28ensure
29  database.release
30end
```
Copy command

> In production, you would query `user_activity` instead.
> The `SummingMergeTree` pre\-merges totals at storage time, keeping reads fast regardless of event volume. The `sum()` in either case handles rows ClickHouse has not yet merged.
