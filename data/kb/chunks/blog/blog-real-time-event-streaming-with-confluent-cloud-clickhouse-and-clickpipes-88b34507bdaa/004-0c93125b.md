To extract this data to our final table, we use a materialized view that exploits the [JSONExtract family of functions](https://clickhouse.com/docs/en/sql-reference/functions/json-functions).

```
1CREATE TABLE blocks
2(
3  `number` UInt32 CODEC(Delta(4), ZSTD(1)),
4  `hash` String,
5  `parent_hash` String,
6  `nonce` String,
7  `sha3_uncles` String,
8  `logs_bloom` String,
9  `transactions_root` String,
10  `state_root` String,
11  `receipts_root` String,
12  `miner` String,
13  `difficulty` Decimal(38, 0),
14  `total_difficulty` Decimal(38, 0),
15  `size` UInt32 CODEC(Delta(4), ZSTD(1)),
16  `extra_data` String,
17  `gas_limit` UInt32 CODEC(Delta(4), ZSTD(1)),
18  `gas_used` UInt32 CODEC(Delta(4), ZSTD(1)),
19  `timestamp` DateTime CODEC(Delta(4), ZSTD(1)),
20  `transaction_count` UInt16,
21  `base_fee_per_gas` UInt64,
22  `withdrawals_root` String,
23  `withdrawals.index` Array(UInt64),
24  `withdrawals.validator_index` Array(Int64),
25  `withdrawals.address` Array(String),
26  `withdrawals.amount` Array(UInt64)
27)
28ENGINE = MergeTree
29ORDER BY timestamp
```
