```
1WITH repos AS
2	(
3    	SELECT if(repo_name = 'yandex/ClickHouse', 'ClickHouse/ClickHouse', repo_name) AS repo_name
4    	FROM github_events
5    	WHERE event_type IN ['ForkEvent', 'WatchEvent']
6    	GROUP BY repo_name
7    	HAVING (countIf(event_type = 'ForkEvent') > 5000) AND (countIf(event_type = 'WatchEvent') > 25000)
8	)
9SELECT
10	rowNumberInAllBlocks() AS position,
11	repo,
12	round(countIf(step = 2) / count(), 3) AS ratio
13FROM
14(
15	SELECT
16    	repo,
17    	windowFunnel(2592000)(created_at, event_type = 'ForkEvent', event_type = 'PullRequestEvent') AS step
18	FROM github_events
19	WHERE ((repo_name IN (repos)) OR (repo_name = 'yandex/ClickHouse')) AND (event_type IN ['ForkEvent', 'PullRequestEvent'])
20	GROUP BY
21    	actor_login,
22    	if(repo_name = 'yandex/ClickHouse', 'ClickHouse/ClickHouse', repo_name) AS repo
23	HAVING step > 0 //ignore users who can raise a PR without forking
24)
25GROUP BY repo
26ORDER BY ratio DESC
27LIMIT 25
```
