```
1WITH 5 as k_val,
2(
3      SELECT max(iteration)
4      FROM centroids
5) AS c_iteration,
6(
7  -- (1) current centroids
8  SELECT groupArray((centroid, position))
9  FROM
10  (
11      SELECT
12          centroid,
13          avgForEachMerge(vector) AS position
14      FROM centroids
15      WHERE iteration = c_iteration AND k = k_val
16      GROUP BY centroid
17  )
18) AS c_centroids,
19(
20  -- (2) previous centroids
21  SELECT groupArray((centroid, position))
22  FROM
23  (
24      SELECT
25          centroid,
26          avgForEachMerge(vector) AS position
27      FROM centroids
28      WHERE iteration = (c_iteration-1) AND k = k_val
29      GROUP BY centroid
30  )
31) AS c_p_centroids
32-- (6) sum differences
33SELECT sum(changed) FROM (
34  SELECT id,
35  -- (3) current centroid for point
36  (arraySort(c -> (c.2), arrayMap(x -> (x.1, L2Distance(x.2, vector)), c_centroids))[1]).1 AS cluster,
37  -- (4) previous centroid for point
38  (arraySort(c -> (c.2), arrayMap(x -> (x.1, L2Distance(x.2, vector)), c_p_centroids))[1]).1 AS cluster_p,
39  -- (5) difference in allocation
40  if(cluster = cluster_p, 0, 1) as changed
41  FROM points
42)
```

## A test dataset \#

The above has been mostly theoretical. Let's see if the above actually works on a real dataset! For this, we'll use a 3m row subset of the popular NYC taxis dataset as the clusters are hopefully relatable. To create and insert the data from S3:

```
1CREATE TABLE trips (
2  trip_id         	UInt32,
3  pickup_datetime 	DateTime,
4  dropoff_datetime	DateTime,
5  pickup_longitude	Nullable(Float64),
6  pickup_latitude 	Nullable(Float64),
7  dropoff_longitude   Nullable(Float64),
8  dropoff_latitude	Nullable(Float64),
9  passenger_count 	UInt8,
10  trip_distance   	Float32,
11  fare_amount     	Float32,
12  extra           	Float32,
13  tip_amount      	Float32,
14  tolls_amount    	Float32,
15  total_amount    	Float32,
16  payment_type    	Enum('CSH' = 1, 'CRE' = 2, 'NOC' = 3, 'DIS' = 4, 'UNK' = 5),
17  pickup_ntaname  	LowCardinality(String),
18  dropoff_ntaname 	LowCardinality(String)
19)
20ENGINE = MergeTree
21ORDER BY (pickup_datetime, dropoff_datetime);
22
23INSERT INTO trips SELECT trip_id, pickup_datetime, dropoff_datetime, pickup_longitude, pickup_latitude, dropoff_longitude, dropoff_latitude, passenger_count, trip_distance, fare_amount, extra, tip_amount, tolls_amount, total_amount, payment_type, pickup_ntaname, dropoff_ntaname
24FROM gcs('https://storage.googleapis.com/clickhouse-public-datasets/nyc-taxi/trips_{0..2}.gz', 'TabSeparatedWithNames');
```
