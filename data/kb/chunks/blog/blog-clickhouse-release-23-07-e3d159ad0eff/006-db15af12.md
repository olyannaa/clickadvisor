```
1from uk_house_price
2filter town == 'LONDON'
3filter postcode1 != ''
4select {  
5  postcode1, street, price
6}
7group postcode1 (     		 
8  sort {-price}
9  take 1
10)		 
11sort {-price}
12take 1..10
13
14WITH table_0 AS
15	(
16    	SELECT
17        	postcode1,
18        	street,
19        	price
20    	FROM uk_house_price
21    	WHERE (town = 'LONDON') AND (postcode1 != '')
22    	ORDER BY
23        	postcode1 ASC,
24        	price DESC
25    	LIMIT 1 BY postcode1
26	)
27SELECT
28	postcode1,
29	street,
30	price

32ORDER BY price DESC
```
