13@retry(wait=wait_random_exponential(min=1, max=60), stop=stop_after_attempt(20))
14def completion_with_backoff(**kwargs):
15   return openai.ChatCompletion.create(**kwargs)
16
17def extract_topics(text):
18   if text == "":
19       return ""
20   messages = [{
21                   "role": "system",
22                   "content": "You are an AI language model trained to extract entities from Hacker News forum comments."},
23               {
24                   "role": "user",
25                   "content": f"From the following text extract the technologies mentioned as a comma separated list with no spaces. Return an empty string if there are no technologies: {text}"
26               }]
27   try:
28       response = completion_with_backoff(model="gpt-3.5-turbo", messages=messages, max_tokens=30, temperature=0,
29                                          request_timeout=request_timeout)
30       return response.choices[0].message.content.strip()
31   except Exception as e:
32       return f"ERROR - {e}"

34for size in sys.stdin:
35   # collect a batch for performance
