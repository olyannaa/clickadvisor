```
10    bool is_var_args = false;
11    llvm::FunctionType * func_type =
12        llvm::FunctionType::get(int64_type, params, is_var_args);
13
14    llvm::Function::LinkageTypes linkage = 
15    llvm::Function::ExternalLinkage;
16    llvm::Function * function =
17        llvm::Function::Create(func_type, linkage, "sum", module);
18
19    /// Get first argument
20    llvm::Value * first_argument = function->args().begin();
21
22    /// Get second argument
23    llvm::Value * second_argument = function->args().begin() + 1;
24
25    /// Create function entry basic block
26    llvm::BasicBlock * entry = 
27    llvm::BasicBlock::Create(context, "entry", function);
28    b.SetInsertPoint(entry);
29
30    /// Sum arguments
31    llvm::Value * sum = b.CreateAdd(first_argument, second_argument);
32
33    /// Return sum result
34    b.CreateRet(sum);
```

Here we create our CHJIT wrapper, call the `compileModule` function and add the `sum` LLVM function (which consists of 1 BasicBlock) into the Module using IRBuilder. Such `sum` function is compiled into this IR:

```

2    entry:
3    %2 = add i64 %0, %1
4    ret i64 %2
```
