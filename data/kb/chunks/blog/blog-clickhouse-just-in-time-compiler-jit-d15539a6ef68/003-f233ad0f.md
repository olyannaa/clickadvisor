1int64_t jitSum(int64_t value_1, int64_t value_2)
2{
3    /// Allocate page with READ + WRITE + EXECUTE permissions
4    char * res = static_cast(mmap(NULL, 4096, PROT_READ | PROT_WRITE | PROT_EXEC,
5            MAP_PRIVATE | MAP_ANON, -1, 0));
6
7    size_t instruction = 0;
8
9    /// lea (%rdi, %rsi, 1), %rax => rax = rdi + rsi * 1
10    res[instruction++] = 0x48;
11    res[instruction++] = 0x8d;
12    res[instruction++] = 0x04;
13    res[instruction++] = 0x37;
14
15    /// retq
16    res[instruction++] = 0xc3;
17
18    using SumFunction = int64_t (*)(int64_t, int64_t);

22}
