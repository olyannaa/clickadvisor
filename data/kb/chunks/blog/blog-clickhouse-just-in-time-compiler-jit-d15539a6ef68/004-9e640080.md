But one question that is not answered yet is how we can interact with our program from JIT code. Consider this example:

```
1void test_function()
2{
3   printf("JIT example\n");
4}
5
6void jitTestFuncCall()
7{
8   /// Allocate page with READ + WRITE + EXECUTE permissions
9   char * res = static_cast<char *>(mmap(NULL, 4096, PROT_READ | PROT_WRITE | PROT_EXEC,
10       MAP_PRIVATE | MAP_ANON, -1, 0));
11
12   size_t instruction = 0;
13
14   /// movabs [pointer_to_test_function], %rax
15   res[instruction++] = 0x48;
16   res[instruction++] = 0xb8;
17
18   ptrdiff_t test_function_raw_ptr = reinterpret_cast<ptrdiff_t>(test_function);
19   memcpy(res + instruction, &test_function_raw_ptr, sizeof(ptrdiff_t));
20   instruction += sizeof(ptrdiff_t);
21
22   /// callq *%rax
23   res[instruction++] = 0xff;
24   res[instruction++] = 0xd0;
25
26   /// retq
27   res[instruction++] = 0xc3;
28
29   using VoidFunction = void (*)(void);
30   VoidFunction func = reinterpret_cast<VoidFunction>(res);
31   func();
32}
33
34int main(int argc, char ** argv)
35{
36   jitTestFuncCall();
37   return 0;
38}
```
Copy command
In this example, we want to call the `test_function` function from JIT code. We do almost the same stuff as in the previous example, but instead of `lea` instruction encoding, we encode `movabs [pointer_to_test_function], %rax` and `callq *%rax` instructions.

We can verify that it works.

```
1$ g++ -O2 jit_example.cpp -o jit_example
2$ ./jit_example
3
4$ JIT example
```

All interaction between JIT code and code in our main program is typically done this way, we pass pointers to functions in JIT code and call it from JIT code.
