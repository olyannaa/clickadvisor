6  %5 = getelementptr inbounds { ptr, ptr }, ptr %1, i64 1
7  %6 = load { ptr, ptr }, ptr %5, align 8
8  %7 = extractvalue { ptr, ptr } %6, 0
9  %8 = getelementptr inbounds { ptr, ptr }, ptr %1, i64 2
10  %9 = load { ptr, ptr }, ptr %8, align 8
11  %10 = extractvalue { ptr, ptr } %9, 0
12  %11 = getelementptr inbounds { ptr, ptr }, ptr %1, i64 3
13  %12 = load { ptr, ptr }, ptr %11, align 8
14  %13 = extractvalue { ptr, ptr } %12, 0
15  %14 = icmp eq i64 %0, 0
16  br i1 %14, label %end, label %loop
17
18end:                                              ; preds = %loop, %entry

20
21loop:                                             ; preds = %loop, %entry

23  %16 = getelementptr i64, ptr %4, i64 %15        /// Adjust first argument ptr using loop counter
24  %17 = load i64, ptr %16, align 8                /// Load first argument data from adjusted ptr
25  %18 = getelementptr i64, ptr %7, i64 %15        /// Adjust second argument ptr using loop counter

33  store i64 %24, ptr %25, align 8                 /// Write result value thought adjusted result ptr
34  %26 = add i64 %15, 1                            /// Increment loop counter
35  %27 = icmp eq i64 %26, %0                       /// Check if loop need to be terminated
