```
1CREATE VIEW
2    data
3AS WITH
4    (select toDate(min(MONTH)) from flight_data) as start_date,
5    (select toDate(max(MONTH)) from flight_data) as end_date
6SELECT
7    AIRLINE,
8    DEPARTURE_AIRPORT,
9    MONTH,
10    toFloat64(log(PASSENGERS)) as Target,
11    assumeNotNull(dateDiff('month', start_date, MONTH) / dateDiff('month', start_date, end_date)) as Trend,
12    if(toMonth(toDate(MONTH)) = 1, 1, 0) as Dummy1,
13    if(toMonth(toDate(MONTH)) = 2, 1, 0) as Dummy2,
14    if(toMonth(toDate(MONTH)) = 3, 1, 0) as Dummy3,
15    if(toMonth(toDate(MONTH)) = 4, 1, 0) as Dummy4,
16    if(toMonth(toDate(MONTH)) = 5, 1, 0) as Dummy5,
17    if(toMonth(toDate(MONTH)) = 6, 1, 0) as Dummy6,
18    if(toMonth(toDate(MONTH)) = 7, 1, 0) as Dummy7,
19    if(toMonth(toDate(MONTH)) = 8, 1, 0) as Dummy8,
20    if(toMonth(toDate(MONTH)) = 9, 1, 0) as Dummy9,
21    if(toMonth(toDate(MONTH)) = 10, 1, 0) as Dummy10,
22    if(toMonth(toDate(MONTH)) = 11, 1, 0) as Dummy11,
23    if(toMonth(toDate(MONTH)) = 12, 1, 0) as Dummy12
24FROM
25    flight_data
26ORDER BY AIRLINE, DEPARTURE_AIRPORT, MONTH
```
