dispatch in aggregate functions. In ClickHouse, if there is `GROUP BY` without keys, for example `SELECT sum(value), avg(value) FROM test_table`, the aggregate functions process data directly in a batch. For the `sum` function, there is the following implementation:

```
1template <typename Value>
2void NO_INLINE addManyImpl(const Value * __restrict ptr, size_t start, size_t end)
3{
4    ptr += start;
5    size_t count = end - start;
6    const auto * end_ptr = ptr + count;
7
8    /// Loop
9    T local_sum{};
10    while (ptr < end_ptr)
11    {
12        Impl::add(local_sum, *ptr);
13        ++ptr;
14    }
15    Impl::add(sum, local_sum);
16}
```

We can now dispatch to the right implementation based on the fastest available CPU instruction set:

```
1template <typename Value>
2void NO_INLINE addMany(const Value * __restrict ptr, size_t start, size_t end)
3{
4#if USE_MULTITARGET_CODE
5 if (isArchSupported(TargetArch::AVX512BW))
6    {
7        addManyImplAVX512BW(ptr, start, end);
8        return;
9    } 
10    else if (isArchSupported(TargetArch::AVX512F))
11    {
12        addManyImplAVX512F(ptr, start, end);
13        return;
14    }
15    else if (isArchSupported(TargetArch::AVX2))
16    {
17        addManyImplAVX2(ptr, start, end);
18        return;
19    }
20    else if (isArchSupported(TargetArch::SSE42))
21    {
22        addManyImplSSE42(ptr, start, end);
23        return;
24    }
25#endif
26
27    addManyImpl(ptr, start, end);
28}
```
