Now let’s take a look at a CPU dispatch optimization in unary functions:

```
1template <typename A, typename Op>
2struct UnaryOperationImpl
3{
4    using ResultType = typename Op::ResultType;
5    using ColVecA = ColumnVectorOrDecimal<A>;
6    using ColVecC = ColumnVectorOrDecimal<ResultType>
7    using ArrayA = typename ColVecA::Container;
8    using ArrayC = typename ColVecC::Container;
9
10    static void vector(const ArrayA & a, ArrayC & c)
11    {
12        /// Loop Op::apply is template for operation
13        size_t size = a.size();
14        for (size_t i = 0; i < size; ++i)
15            c[i] = Op::apply(a[i]);
16    }
17
18    static void constant(A a, ResultType & c)

20        c = Op::apply(a);
21    }
22};
