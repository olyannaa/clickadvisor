```
1Our query here:
2WITH
3   1024 AS height, 1024 AS width, 6 AS resolution, 30 AS lc, 144 AS uc,
4   h3_rates AS
5   (
6       SELECT
7           geoToH3(centroid.1, centroid.2, resolution) AS h3,
8           avg(avg_d_kbps) AS download_speed
9       FROM ookla
10       – filter out as these are sporadic and don’t cover the continent
11       WHERE dictGet(country_polygons, 'name', centroid) != 'Antarctica'
12       GROUP BY h3
13   ),
14   (
15       SELECT min(download_speed) FROM h3_rates
16   ) AS min_speed,
17   (
18       SELECT quantile(0.95)(download_speed) FROM h3_rates
19   ) AS max_speed,
20   h3_colors AS
21   (
22       SELECT
23           -- sqrt gradient
24           arrayMap(x -> (x.2), arrayConcat(h3ToGeoBoundary(h3), [h3ToGeoBoundary(h3)[1]])) AS longs,
25           sqrt(download_speed - min_speed) / sqrt(max_speed - min_speed) AS percent,
26           -- oklch color with gradient on hue
27           format('oklch(60% 0.23 {})', toUInt8(lc + (percent * (uc - lc)))) AS oklch,
28           arrayMap(p -> mercator((p.2, p.1), width, height), h3ToGeoBoundary(h3)) AS pixel_poly
29       FROM h3_rates
30       -- filter out points crossing 180 meridian
31       WHERE arrayExists(i -> (abs((longs[i]) - (longs[i + 1])) > 180), range(1, length(longs) + 1)) = 0
32   )
33SELECT SVG(pixel_poly, format('fill: {};', oklch))
34FROM h3_colors
35INTO OUTFILE 'h3_countries.svg' TRUNCATE
36FORMAT CustomSeparated
37SETTINGS format_custom_result_before_delimiter = '<svg style="background-color: rgb(17,17,17);" viewBox="0 0 1024 1024" xmlns:svg="http://www.w3.org/2000/svg" xmlns="http://www.w3.org/2000/svg">', format_custom_result_after_delimiter = '</svg>', format_custom_escaping_rule = 'Raw'
38
3955889 rows in set. Elapsed: 10.101 sec. Processed 384.02 million rows, 8.06 GB (38.02 million rows/s., 798.38 MB/s.)
40Peak memory usage: 23.73 MiB.
```
