```
1CREATE OR REPLACE VIEW ookla_as_pixels AS
2WITH [234, 36, 19] AS lc, [0, 156, 31] AS uc, [17,17,17] As bg,
3pixels AS
4(
5	SELECT (y * 2048) + x AS pos,
6    	avg(avg_d_kbps) AS download_speed
7	FROM ookla
8	GROUP BY
9    	round(mercator(centroid, 2048, 2048).1) AS x,
10    	round(mercator(centroid, 2048, 2048).2) AS y
11	ORDER BY
12    	pos WITH FILL FROM 0 TO 2048 * 2048
13),
14(
15	SELECT min(download_speed)
16	FROM pixels
17) AS min_speed,
18(
19	SELECT quantile(0.95)(download_speed)
20	FROM pixels
21) AS max_speed,
22pixel_colors AS (
23	SELECT
24	least(if(isNull(download_speed),0, (download_speed - min_speed) / (max_speed - min_speed)), 1.0) AS percent,
25	if(isNull(download_speed),bg[1], toUInt8(lc[1] + (percent * (uc[1] - lc[1])))) AS red,
26	if(isNull(download_speed),bg[2], toUInt8(lc[2] + (percent * (uc[2] - lc[2])))) AS green,
27	if(isNull(download_speed),bg[3], toUInt8(lc[3] + (percent * (uc[3] - lc[3])))) AS blue
28	FROM pixels
29) SELECT red::UInt8, green::UInt8, blue::UInt8, 255::UInt8 as alpha FROM pixel_colors
```

Converting this data to an image is most easily done using a [Canvas element](https://www.w3schools.com/html/html5_canvas.asp) and some simple js. We effectively have 4 bytes per pixel and 2048 x 2048 x 4 bytes in total or approximately 16MB. Using the `putImageData` we render this data directly into a canvas.
