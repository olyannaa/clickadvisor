SVG. ![country_svg.png](/_next/image?url=%2Fuploads%2Fcountry_svg_03800935c7.png&w=2048&q=75) The above SVG is 9\.2MB and is generated by the following query, which completes in 11 seconds. As shown in the key, red here represents slow internet speed (download rate), while green represents faster connection speed.

```
1WITH
2   -- settings determining height, width and color range
3   1024 AS height, 1024 AS width, [234, 36, 19] AS lc, [0, 156, 31] AS uc,
4   country_rates AS
5   (
6       -- compute the average download speed per country using country_polygons dictionary
7       SELECT
8           dictGet(country_polygons, 'name', centroid) AS country,
9           avg(avg_d_kbps) AS download_speed
10       FROM ookla
11       -- exclude 'Antarctica' as polygon isnt fully closed
12       WHERE country != 'Antarctica'
13       GROUP BY country
14   ),
15   (
16       -- compute min speed over all countries
17       SELECT min(download_speed)
18       FROM country_rates
19   ) AS min_speed,
20   (
21       -- compute max speed over all countries
22       SELECT max(download_speed)
23       FROM country_rates
24   ) AS max_speed,
25   country_colors AS
26   (
27       SELECT
28           country,
29           download_speed,
30           (download_speed - min_speed) / (max_speed - min_speed) AS percent,
31           -- compute a rgb value based on linear interolation between yellow and red
32           format('rgb({},{},{})', toUInt8((lc[1]) + (percent * ((uc[1]) - (lc[1])))), toUInt8((lc[2]) + (percent * ((uc[2]) - (lc[2])))), toUInt8((lc[3]) + (percent * ((uc[3]) - (lc[3]))))) AS rgb,
33           --  get polygon for country name and mercator project every point
34           arrayMap(p -> arrayMap(r -> arrayMap(x -> mercator(x, width, height), r), p), dictGet(country_names, 'coordinates', country)) AS pixel_poly
35       FROM country_rates
36   )
37-- convert country polygon to svg
38SELECT SVG(pixel_poly, format('fill: {};', rgb))
39FROM country_colors
40-- send output to file
41INTO OUTFILE 'countries.svg'
42FORMAT CustomSeparated
43SETTINGS format_custom_result_before_delimiter = '<svg style="background-color: rgb(17,17,17);" viewBox="0 0 1024 1024" xmlns:svg="http://www.w3.org/2000/svg" xmlns="http://www.w3.org/2000/svg">', format_custom_result_after_delimiter = '</svg>', format_custom_escaping_rule = 'Raw'
44
45246 rows in set. Elapsed: 1.396 sec. Processed 384.02 million rows, 8.06 GB (275.15 million rows/s., 5.78 GB/s.)
46Peak memory usage: 39.54 MiB.
```
