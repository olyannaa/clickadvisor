```
1WITH all_cells AS (
2      SELECT
3          assumeNotNull(geoToH3(pickup_latitude, pickup_longitude, 9)) AS cell,
4          count() AS trips
5      FROM trips_small
6      WHERE pickup_latitude BETWEEN 40.4 AND 40.95
7        AND pickup_longitude BETWEEN -74.3 AND -73.7
8      GROUP BY cell
9      HAVING trips >= 10
10  ),
11  max_log AS (SELECT max(log(trips)) AS max_log FROM all_cells),
12  top_cells AS (
13      SELECT pickup_h3 AS cell, count() AS trips
14      FROM trips_small_h3
15      WHERE pickup_h3 != 0
16      GROUP BY cell
17      HAVING h3ToGeo(cell).latitude BETWEEN 40.0 AND 42.0
18      ORDER BY trips DESC
19      LIMIT 10
20  ),
21  background AS (
22      SELECT replaceAll(
23          svg([arrayMap(p -> (p.2, p.1), h3ToGeoBoundary(cell))]::Polygon),
24          'style=""',
25          concat('style="fill:rgba(250,255,105,', toString(round(log(trips) / max_log, 3)), ');stroke:none"')
26      ) AS fragment
27      FROM all_cells, max_log
28  ),
29  highlights AS (
30      SELECT replaceAll(
31          svg([arrayMap(p -> (p.2, p.1), h3ToGeoBoundary(cell))]::Polygon),
32          'style=""',
33          'style="fill:rgba(255,255,255,0.9);stroke:#161517;stroke-width:0.0005"'
34      ) AS fragment
35      FROM top_cells
36  )
37  SELECT concat(
38      '<svg xmlns="http://www.w3.org/2000/svg" viewBox="-74.05 40.58 0.32 0.35" width="650" height="1000" preserveAspectRatio="none">',
39      '<rect x="-74.05" y="40.58" width="0.32" height="0.35" fill="#161517"/>',
40      '<g transform="scale(1,-1) translate(0,-81.51)">',
41      arrayStringConcat((SELECT groupArray(fragment) FROM background), ''),
42      arrayStringConcat((SELECT groupArray(fragment) FROM highlights), ''),
43      '</g></svg>'
44  )
45  FORMAT TSVRaw;
```

For a radius query \- all pickups within two H3 cells of Times Square \- `h3kRing` gives us the set of cells to look up, and the primary key index does the rest:

```
1SELECT count() AS trips
2FROM trips_small_h3
3WHERE pickup_h3 IN (
4    SELECT arrayJoin(h3kRing(geoToH3(40.758, -73.985, 9), 2))
5);
```
