While similar, this join becomes a little more complex (and costly) if using "per entity" feature tables. Assuming we have the feature tables `domain_features` and `user_features` populated with [these queries](https://pastila.nl/?00226ff6/666bf2f425b877ae8cdf08bf2dfa3ee5#0q0sxENDDPzchVE7DZZJRA==).

```
1INSERT INTO predict_bounce_model SELECT
2   rand() AS Row,
3   mt.EventTime AS EventTime,
4   mt.UserID AS UserID,
5   mt.Domain AS Domain,
6   mt.UserAgent,
7   mt.RefererCategoryID,
8   mt.URLCategoryID,
9   DomainsVisitedPerHour,
10   UniqueIPsPerHour,
11   AverageRequestTime,
12   mt.IsNotBounce
13FROM predict_bounce_subset AS mt FINAL
14ASOF LEFT JOIN (
15   SELECT Domain, EventTime, Value.UInt64 AS UniqueIPsPerHour
16   FROM domain_features
17   WHERE FeatureId = 'number_unique_ips_per_hour'
18) AS df ON (mt.Domain = df.Domain) AND (mt.EventTime >= df.EventTime)
19ASOF LEFT JOIN (
20   SELECT Domain, EventTime, Value.Float64 AS AverageRequestTime
21   FROM domain_features
22   WHERE FeatureId = 'average_request_time'
23) AS art ON (mt.Domain = art.Domain) AND (mt.EventTime >= art.EventTime)
24ASOF LEFT JOIN (
25   SELECT UserID, EventTime, Value.UInt64 AS DomainsVisitedPerHour
26   FROM user_features
27   WHERE FeatureId = 'domains_visited_per_hour'
28) AS dv ON (mt.UserID = dv.UserID) AND (mt.EventTime >= dv.EventTime)
```
